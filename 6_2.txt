using System;

public class Zadanie
{
    public static void Main(string[] args)
    {
        int[] tablica=new int[6];
        int suma=0;
        double srednia=0;
        int wiekszeOdSredniej=0;

        for(int i=0;i<6;i++){
            Console.Write("Podaj "+(i+1)+" liczbę: ");
            tablica[i]=int.Parse(Console.ReadLine());
            suma+=tablica[i];
        }

        srednia=suma/6;
        for(int i=0;i<6;i++){
            if(tablica[i]>srednia){
                wiekszeOdSredniej++;
            }
        }

        Console.WriteLine("Suma wszystkich liczb: "+suma);
        Console.WriteLine("Średnia arytmetyczna: "+srednia);
        Console.WriteLine("Liczba elementów większych od średniej: "+wiekszeOdSredniej);
    }
}
